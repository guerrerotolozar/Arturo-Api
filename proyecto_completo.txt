/**
 * Controlador de IA — Consulta RAG + Web híbrida.
 * Ruta: POST /IA/consultar
 * Motor: llama3.1 (DEEP) — Procesamiento profundo de contexto.
 */

const { necesitaInternet, consultarIA } = require('../services/llmService');
const { buscarContexto } = require('../services/vaultService');
const { buscarEnWeb } = require('../services/searchService');
const { obtenerFechaHoraLocal } = require('../utils/dateUtils');

/**
 * Handler para POST /IA/consultar
 * Orquesta: intención → vault → web → LLM → respuesta.
 */
async function consultaIA(req, res) {
    const { pregunta } = req.body;

    if (!pregunta || pregunta.trim().length === 0) {
        return res.status(400).json({
            error: 'Falta la pregunta, Ronald. Envía { "pregunta": "Tu pregunta aquí" }'
        });
    }

    console.log(`[IA] Consulta recibida: "${pregunta}"`);

    try {
        // PASO 1: Detectar intención — ¿Necesita internet? (con llama3.1)
        const quiereWeb = await necesitaInternet(pregunta);

        // PASO 2: Buscar contexto en el vault (siempre)
        const fragmentosVault = buscarContexto(pregunta);
        console.log(`[IA] ${fragmentosVault.length} fragmentos del vault encontrados`);

        // PASO 3: Buscar en internet (si Arturo lo decidió)
        let datosWeb = null;
        if (quiereWeb) {
            try {
                datosWeb = await buscarEnWeb(pregunta);
                console.log(`[IA] Datos web obtenidos: ${datosWeb.results.length} resultados`);
            } catch (webErr) {
                console.warn(`[IA] Búsqueda web falló: ${webErr.message}. Continuando sin web.`);
            }
        }

        // PASO 4: Si no hay contexto de ningún lado, avisar
        if (fragmentosVault.length === 0 && !datosWeb) {
            return res.status(200).json({
                pregunta: pregunta,
                respuesta: 'No encontré notas relevantes en tu vault ni pude obtener datos de internet para responder esa pregunta. ¿Quieres que cree una nota nueva sobre ese tema?',
                fuentes_vault: [],
                fuentes_web: [],
                contexto_encontrado: false,
                busco_en_web: quiereWeb
            });
        }

        // PASO 5: Consultar a Ollama con el contexto híbrido usando modelo profundo
        const fechaHora = obtenerFechaHoraLocal();
        const respuesta = await consultarIA(pregunta, fragmentosVault, datosWeb, fechaHora);

        // PASO 6: Respuesta estructurada
        res.status(200).json({
            pregunta: pregunta,
            respuesta: respuesta,
            fuentes_vault: fragmentosVault.map(f => ({
                nota: f.fuente,
                relevancia: f.score
            })),
            fuentes_web: datosWeb ? datosWeb.results.map(r => ({
                titulo: r.title,
                url: r.url
            })) : [],
            contexto_encontrado: true,
            busco_en_web: quiereWeb,
            fragmentos_vault_usados: fragmentosVault.length,
            resultados_web_usados: datosWeb ? datosWeb.results.length : 0,
            timestamp: fechaHora
        });

        console.log(`[IA] Respuesta entregada exitosamente (web: ${quiereWeb})`);

    } catch (err) {
        console.error(`[IA ERROR] ${err.message}`);

        // Diferenciar errores de Ollama vs Tavily vs otros
        if (err.message.includes('Ollama') || err.message.includes('fetch')) {
            return res.status(503).json({
                error: 'Ollama no está respondiendo. ¿Está corriendo el servicio?',
                detalle: err.message
            });
        }

        if (err.message.includes('Tavily')) {
            return res.status(503).json({
                error: 'Tavily no está respondiendo. ¿La API key está configurada?',
                detalle: err.message
            });
        }

        res.status(500).json({
            error: 'Algo falló internamente, Ronald.',
            detalle: err.message
        });
    }
}

module.exports = consultaIA;

/**
 * Controlador de IA Fast — Ahora funciona como AGENTE AUTÓNOMO.
 * Motor: qwen2.5:1.5B (Orquestador)
 */

const { orquestarTareas, consultarIAFast, llamarOllama } = require('../services/llmService');
const { buscarEnWeb } = require('../services/searchService');
const { buscarContexto } = require('../services/vaultService');

async function consultaIAFast(req, res) {
    const { pregunta } = req.body;
    if (!pregunta) return res.status(400).send('Falta la pregunta rápida.');

    console.log(`\n[AGENTE] Analizando petición: "${pregunta}"`);

    try {
        // 1. Qwen piensa y decide qué herramientas usar (Retorna JSON)
        const plan = await orquestarTareas(pregunta);
        console.log(`[AGENTE] Plan de Qwen:`, JSON.stringify(plan.tareas));

        let respuestaFinal = "";
        let herramientasUsadas = [];

        // 2. JavaScript ejecuta las herramientas de forma determinista (sin alucinar)
        for (const tarea of plan.tareas) {
            
            if (tarea.herramienta === 'calculadora') {
                try {
                    // Usamos el motor V8 de Node para calcular, cero errores matemáticos
                    // Reemplazamos 'x' por '*' y '^' por '**' para que JS entienda
                    const formulaLimpia = tarea.parametros.formula.replace(/x/gi, '*').replace(/\^/g, '**');
                    const resultadoMath = new Function('return ' + formulaLimpia)();
                    
                    respuestaFinal += `El resultado de la operación es: ${resultadoMath}\n`;
                    herramientasUsadas.push('Calculadora JS');
                    console.log(`[TOOL] Calculadora ejecutada: ${formulaLimpia} = ${resultadoMath}`);
                } catch (e) {
                    respuestaFinal += `Error calculando la fórmula: ${tarea.parametros.formula}\n`;
                }
            } 
            
            else if (tarea.herramienta === 'tavily_search') {
                const web = await buscarEnWeb(tarea.parametros.query);
                const resumen = web.answer || (web.results[0] ? web.results[0].content : 'Sin datos');
                respuestaFinal += `Datos de internet: ${resumen}\n`;
                herramientasUsadas.push('Tavily Search');
            }
            
            else if (tarea.herramienta === 'vault_search') {
                console.log(`[TOOL] Leyendo el Cerebro de Ronald buscando: "${tarea.parametros.query}"`);
                const fragmentos = buscarContexto(tarea.parametros.query);
                
                let resumenVault = "Datos encontrados en mis notas locales:\n";
                if (fragmentos.length > 0) {
                    fragmentos.forEach(f => {
                        // Le pasamos la fuente y el pedazo de texto a Qwen
                        resumenVault += `- Fuente [${f.fuente}]: ${f.texto}\n`;
                    });
                } else {
                    resumenVault += "No encontré nada escrito sobre esto en mi base de datos.\n";
                }
                
                respuestaFinal += resumenVault;
                herramientasUsadas.push('Vault Search');
            }
            
            else {
                // Fallback: Si es solo charla, le responde normal
                const chat = await consultarIAFast(pregunta);
                respuestaFinal += chat;
                herramientasUsadas.push('Chat Directo');
            }
        }

      // 3. Síntesis de la IA ---
        if (herramientasUsadas.includes('Tavily Search') || herramientasUsadas.includes('Calculadora JS') || herramientasUsadas.includes('Vault Search')) {
            console.log(`[AGENTE] Traduciendo y adaptando respuesta para Ronald...`);
            
            const promptSintesis = `Traduce y adapta la siguiente información para responderle a Ronald.

        REGLAS ESTRICTAS:
        1. DEBES responder EXCLUSIVAMENTE en español. Cero inglés.
        2. Usa un tono de asistente cercano, directo y con jerga local colombiana (ej. "Parce", "Pille", "Listo").
        3. Solo entrega la respuesta final, no digas "Aquí tienes la traducción" ni expliques qué hiciste.
        Datos crudos que debes comunicar: 
        "${respuestaFinal.trim()}"

        Lo que preguntó Ronald: 
        "${pregunta}"

        Tu respuesta directa en español:`;
                        
            respuestaFinal = await llamarOllama(promptSintesis, "qwen2.5:1.5b");
        }

        // 4. Devolvemos el resultado a Postman/Frontend
        res.status(200).json({
            pregunta: pregunta,
            respuesta: respuestaFinal.trim(),
            motor_usado: "qwen2.5:1.5b (Modo Agente)",
            herramientas_activadas: herramientasUsadas,
            plan_interno: plan.tareas
        });

    } catch (err) {
        console.error(`[AGENTE ERROR] ${err.message}`);
        res.status(500).json({ error: 'Fallo en la ejecución del agente.' });
    }
}

module.exports = consultaIAFast;

/**
 * Controlador del Webhook del Segundo Cerebro.
 * Ruta: POST /webhook/cerebro
 * Motor: phi3 (FAST) — Inserción rápida de ideas al vault.
 */

const { clasificarIdea } = require('../services/llmService');
const { obtenerCategorias, escribirIdeaEnVault } = require('../services/vaultService');
const { obtenerTimestamp } = require('../utils/dateUtils');

/**
 * Handler para POST /webhook/cerebro
 * Recibe una idea, responde inmediato, y clasifica + guarda en segundo plano.
 */
function webhookCerebro(req, res) {
    const { idea } = req.body;

    if (!idea) return res.status(400).send('Falta la idea, Ronald.');

    // Responder de inmediato (evita timeout en Postman)
    res.status(200).json({
        mensaje: "Idea recibida. Arturo está procesando...",
        idea_original: idea
    });

    // Procesar en segundo plano con phi3 vía API HTTP
    console.log(`[LOG] Procesando idea: "${idea}"`);

    const categorias = obtenerCategorias();

    clasificarIdea(idea, categorias)
        .then(categoria => {
            const timestamp = obtenerTimestamp();
            escribirIdeaEnVault(categoria, idea, timestamp);
            console.log(`[SUCCESS] Idea guardada en [[${categoria}]]`);
        })
        .catch(err => {
            console.error(`[ERROR] Fallo al clasificar: ${err.message}`);
        });
}

module.exports = webhookCerebro;

/**

 * Body esperado:

 *   "titulo":   "Nombre de la nota",          // requerido
 *   "contexto": "Descripción o prompt breve", // requerido — IA expande esto
 *   "carpeta":  "Finanzas",                   // opcional (default: Inbox)
 *   "tipo":     "proyecto"                    // opcional (default: nota)
 * }
 */

const { consultarIAFast } = require('../services/llmService');
const { crearNotaEstructurada } = require('../services/vaultService');
const { obtenerTimestamp, obtenerFechaHoraLocal } = require('../utils/dateUtils');

/**
 * Handler para POST /IA/nota
 */
async function crearNota(req, res) {
    const { titulo, contexto, carpeta = 'Inbox', tipo = 'nota' } = req.body;

    if (!titulo || !contexto) {
        return res.status(400).json({
            error: 'Faltan campos requeridos.',
            requeridos: { titulo: 'string', contexto: 'string' },
            opcionales: { carpeta: 'string (default: Inbox)', tipo: 'string (default: nota)' }
        });
    }

    console.log(`[IA NOTA] Generando nota: "${titulo}" → ${carpeta}/`);

    try {
        // IA genera el cuerpo de la nota basándose en el contexto
        const promptContenido = `Eres IA, el asistente personal de Ronald, un desarrollador Full-Stack colombiano.
Tu tarea es generar el contenido Markdown para una nota de su Segundo Cerebro en Obsidian.

Instrucciones:
- Escribe SOLO el contenido Markdown (sin el título H1, ese ya se agrega automáticamente).
- Usa formato Obsidian: [[Concepto]] para enlazar entidades, metas, proyectos o tecnologías.
- Estructura con secciones H2 (##) cuando tenga sentido.
- Tono técnico, ejecutivo, directo. Sin relleno.
- Máximo 400 palabras.

Título de la nota: "${titulo}"
Tipo: ${tipo}
Carpeta destino: ${carpeta}
Contexto / instrucciones específicas: ${contexto}

Contenido Markdown:`;

        const contenidoGenerado = await consultarIAFast(promptContenido);

        const timestamp = obtenerTimestamp();
        const resultado = crearNotaEstructurada({
            titulo,
            contenido: '\n' + contenidoGenerado,
            carpeta,
            tipo,
            timestamp
        });

        if (!resultado.creada) {
            return res.status(409).json({
                mensaje: `La nota "${titulo}" ya existe en ${carpeta}/. No se sobreescribió.`,
                ruta: resultado.ruta
            });
        }

        res.status(201).json({
            mensaje: `Nota creada correctamente.`,
            ruta: resultado.ruta,
            titulo,
            carpeta,
            tipo,
            timestamp,
            contenido_preview: contenidoGenerado.substring(0, 200) + '...'
        });

        console.log(`[IA NOTA] Nota creada: ${resultado.ruta}`);

    } catch (err) {
        console.error(`[IA NOTA ERROR] ${err.message}`);
        res.status(500).json({ error: 'Fallo al crear la nota.', detalle: err.message });
    }
}

module.exports = crearNota;

/**
 * Devuelve:
 *   - Estado de Ollama y modelos cargados
 *   - Conteo de notas en el vault
 *   - Estado de Tavily (si la key está configurada)
 *   - Uptime del proceso Node.js
 */

const { OLLAMA_URL, TAVILY_API_KEY, VAULT_PATH, OLLAMA_MODEL_FAST, OLLAMA_MODEL_DEEP } = require('../config/env');
const { leerVaultRecursivo } = require('../services/vaultService');

/**
 * Consulta los modelos disponibles en Ollama.
 * Usa el endpoint /api/tags (estándar de Ollama).
 * @returns {Promise<string[]>} - Lista de nombres de modelos
 */
async function obtenerModelosOllama() {
    const baseUrl = OLLAMA_URL.replace('/api/generate', '');
    const response = await fetch(`${baseUrl}/api/tags`, { method: 'GET' });
    if (!response.ok) throw new Error(`Ollama /api/tags respondió ${response.status}`);
    const data = await response.json();
    return (data.models || []).map(m => m.name);
}

/**
 * Handler para GET /api/status
 */
async function obtenerStatus(req, res) {
    const inicio = Date.now();

    // — Vault —
    let vault = { notas: 0, carpetas: new Set(), error: null };
    try {
        const notas = leerVaultRecursivo();
        vault.notas = notas.length;

        const path = require('path');
        notas.forEach(n => {
            const dir = path.dirname(n.ruta).replace(/\\/g, '/').replace(/^\//, '');
            if (dir && dir !== '.') vault.carpetas.add(dir.split('/')[0]);
        });
    } catch (err) {
        vault.error = err.message;
    }

    // — Ollama —
    let ollama = { ok: false, modelos: [], modeloFast: OLLAMA_MODEL_FAST, modeloDeep: OLLAMA_MODEL_DEEP, error: null };
    try {
        ollama.modelos = await obtenerModelosOllama();
        ollama.ok      = true;

        const fastDisponible = ollama.modelos.some(m => m.includes(OLLAMA_MODEL_FAST));
        const deepDisponible = ollama.modelos.some(m => m.includes(OLLAMA_MODEL_DEEP));

        if (!fastDisponible) console.warn(`[STATUS] Modelo FAST "${OLLAMA_MODEL_FAST}" no encontrado en Ollama.`);
        if (!deepDisponible) console.warn(`[STATUS] Modelo DEEP "${OLLAMA_MODEL_DEEP}" no encontrado en Ollama.`);

        ollama.fastDisponible = fastDisponible;
        ollama.deepDisponible = deepDisponible;
    } catch (err) {
        ollama.error = err.message;
    }

    // — Tavily —
    const tavily = {
        configurada: !!TAVILY_API_KEY,
        nota: TAVILY_API_KEY ? 'API Key presente en .env' : 'Sin API Key — búsqueda web deshabilitada'
    };

    // — Uptime —
    const uptimeSegundos = Math.floor(process.uptime());
    const horas   = Math.floor(uptimeSegundos / 3600);
    const minutos = Math.floor((uptimeSegundos % 3600) / 60);
    const segundos = uptimeSegundos % 60;
    const uptime  = `${horas}h ${minutos}m ${segundos}s`;

    const latencia = Date.now() - inicio;

    res.status(200).json({
        servidor:  'Arturo',
        estado:    ollama.ok ? 'operativo' : 'degradado',
        timestamp: new Date().toLocaleString('es-CO'),
        latencia_ms: latencia,
        uptime,
        vault: {
            notas:    vault.notas,
            carpetas: vault.carpetas.size,
            path:     VAULT_PATH,
            error:    vault.error
        },
        ollama,
        tavily
    });
}

// PROBLEMA: Phi-3 fuerza [[Conceptos]] y parrafos largos incluso en preguntas
// triviales ("2+2"), tardando 14s generando texto basura.

const PATRONES_PREGUNTA_SIMPLE = [
    // Matematicas puras
    /^[\d\s\+\-\*\/\(\)\^%=.,]+\??\s*$/,
    /cuanto es\s+[\d\s\+\-\*\/]+/i,
    /cuanto\s+(son|da|seria)\s+[\d]/i,
    /\d+\s*[\+\-\*\/x\u00f7]\s*\d+/,

    /^(hola|buenas|hey|que hubo|quiubo|ey|hi)[^a-z]*$/i,
    /^(gracias|listo|ok|okay|perfecto|dale)[^a-z]*$/i,
    /^.?(como estas|como vas|que tal)[^a-z]*\??$/i,
    // Datos concretos de una sola dimension
    /^.?cual es (el|la) (capital|moneda|idioma) de/i,
    /^.?en que ano (nacio|murio|fue)/i,
];

/**
 * Evalua si una pregunta es simple (respuesta directa en 1-2 lineas).
 * @param {string} pregunta
 * @returns {boolean}
 */
function esPreguntaSimple(pregunta) {
    const texto = pregunta.trim();

    if (PATRONES_PREGUNTA_SIMPLE.some(p => p.test(texto))) return true;

    // Heuristica de longitud + ausencia de palabras de analisis profundo
    const palabras = texto.split(/\s+/).filter(Boolean);
    const triggersProfundos = ['explica', 'explicame', 'como funciona', 'por que', 'diferencia entre', 'ventaja', 'describe', 'analiza', 'resume'];
    const tieneContextoComplejo = triggersProfundos.some(p => texto.toLowerCase().includes(p));

    if (palabras.length <= 7 && !tieneContextoComplejo) return true;

    return false;
}

 * Los ojos de Arturo en internet.

const { TAVILY_API_KEY, TAVILY_URL } = require('../config/env');

/**
 * Busca información en internet usando la API de Tavily.
 * Usa search_depth 'advanced' para resultados más precisos.
 * @param {string} query - Lo que se va a buscar
 * @returns {Promise<{answer: string|null, results: Array<{title: string, url: string, content: string}>}>}
 */
async function buscarEnWeb(query) {
    console.log(`[TAVILY] Buscando: "${query}"`);

    const response = await fetch(TAVILY_URL, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({
            api_key: TAVILY_API_KEY,
            query: query,
            search_depth: 'advanced',
            include_answer: true,
            max_results: 5
        })
    });

    if (!response.ok) {
        const errorBody = await response.text().catch(() => 'Sin detalle');
        throw new Error(`Tavily respondió con status ${response.status}: ${errorBody}`);
    }

    const data = await response.json();

    const resultados = (data.results || []).map(r => ({
        title: r.title || 'Sin título',
        url: r.url || '',
        content: (r.content || '').substring(0, 600)
    }));

    console.log(`[TAVILY] ${resultados.length} resultados obtenidos`);

    return {
        answer: data.answer || null,
        results: resultados
    };
}

const fs = require('fs');
const path = require('path');
const { VAULT_PATH } = require('../config/env');

// ============================================================

/**
 * Garantiza que un directorio exista, creándolo recursivamente si hace falta.
 * @param {string} dirPath
 */
function asegurarDirectorio(dirPath) {
    if (!fs.existsSync(dirPath)) {
        fs.mkdirSync(dirPath, { recursive: true });
    }
}

// ============================================================
// LECTURA DEL VAULT
// ============================================================

/**
 * Lee recursivamente todos los archivos .md dentro del vault.
 * Ignora carpetas ocultas (como .obsidian).
 * @param {string} dir
 * @returns {Array<{ruta: string, nombre: string, contenido: string}>}
 */
function leerVaultRecursivo(dir = VAULT_PATH) {
    const resultados = [];

    try {
        const entradas = fs.readdirSync(dir, { withFileTypes: true });

        for (const entrada of entradas) {
            const rutaCompleta = path.join(dir, entrada.name);

            if (entrada.isDirectory()) {
                if (entrada.name.startsWith('.')) continue;
                resultados.push(...leerVaultRecursivo(rutaCompleta));
            } else if (entrada.name.endsWith('.md')) {
                try {
                    const contenido = fs.readFileSync(rutaCompleta, 'utf-8');
                    resultados.push({
                        ruta: rutaCompleta.replace(VAULT_PATH, ''),
                        nombre: path.basename(rutaCompleta, '.md'),
                        contenido
                    });
                } catch (err) {
                    console.error(`[WARN] No se pudo leer ${rutaCompleta}: ${err.message}`);
                }
            }
        }
    } catch (err) {
        console.error(`[WARN] No se pudo acceder al directorio ${dir}: ${err.message}`);
    }

    return resultados;
}

/**
 * Escribe una idea en la nota correcta del vault.
 *   1. Busca la nota por nombre recursivamente (respeta reorganizaciones de carpetas).
 *   2. Si la nota no existe, la crea en /Inbox/ con frontmatter YAML básico.
 *   3. Si la IA pasó una rutaEspecifica, la respeta por encima de todo.
 *
 * @param {string} categoria   - Nombre de la nota destino (sin extensión)
 * @param {string} idea        - Texto de la idea
 * @param {string} timestamp   - Timestamp formateado
 * @param {string|null} rutaEspecifica - Ruta relativa opcional (ej: 'Finanzas/NuevaCategoria.md')
 */
function escribirIdeaEnVault(categoria, idea, timestamp, rutaEspecifica = null) {
    let archivoDestino;

    if (rutaEspecifica) {
        // La IA especificó una ruta concreta — respetar al 100%
        archivoDestino = path.join(VAULT_PATH, rutaEspecifica);
        asegurarDirectorio(path.dirname(archivoDestino));

    } else {
        // Buscar la nota real en el vault (recursivo)
        const notas = leerVaultRecursivo();
        const notaReal = notas.find(n =>
            n.nombre.toLowerCase() === categoria.toLowerCase()
        );

        if (notaReal) {
            // Nota encontrada en su ruta actual — appendear ahí
            archivoDestino = path.join(VAULT_PATH, notaReal.ruta);

        } else {
            // Nota nueva — crear en /Inbox/ con frontmatter
            const inboxDir = path.join(VAULT_PATH, 'Inbox');
            asegurarDirectorio(inboxDir);
            archivoDestino = path.join(inboxDir, `${categoria}.md`);

            if (!fs.existsSync(archivoDestino)) {
                const frontmatter = [
                    '---',
                    `tipo: categoria_nueva`,
                    `creado: ${timestamp}`,
                    `fuente: webhook_cerebro`,
                    `revisado: false`,
                    '---',
                    `# ${categoria}`,
                    '',
                    '> Nota creada automáticamente por Arturo. Mover a la carpeta correcta cuando sea relevante.',
                    ''
                ].join('\n');
                fs.writeFileSync(archivoDestino, frontmatter);
                console.log(`[INBOX] Nueva categoría creada: Inbox/${categoria}.md`);
            }
        }
    }

    const entrada = `\n- [${timestamp}] ${idea}`;
    fs.appendFileSync(archivoDestino, entrada);
    console.log(`[VAULT] Idea guardada en: ${archivoDestino.replace(VAULT_PATH, '')}`);
}

/**
 * Crea una nota .md completa con frontmatter, título y contenido.
 * Si la ruta ya existe, lanza un error (no sobreescribe sin aviso).
 *
 * @param {object} params
 * @param {string} params.titulo     - Título de la nota (también nombre del archivo)
 * @param {string} params.contenido  - Cuerpo en Markdown
 * @param {string} params.carpeta    - Subcarpeta relativa dentro del vault (ej: 'Finanzas')
 * @param {string} params.tipo       - Valor del campo `tipo:` en el frontmatter
 * @param {string} params.timestamp  - Timestamp de creación
 * @returns {{ ruta: string, creada: boolean }}
 */
function crearNotaEstructurada({ titulo, contenido, carpeta = 'Inbox', tipo = 'nota', timestamp }) {
    const nombreArchivo = titulo.replace(/[^a-zA-ZáéíóúÁÉÍÓÚñÑ0-9\s_-]/g, '').trim().replace(/\s+/g, '_');
    const dirDestino = path.join(VAULT_PATH, carpeta);

    asegurarDirectorio(dirDestino);

    const archivoDestino = path.join(dirDestino, `${nombreArchivo}.md`);
    const yaExiste = fs.existsSync(archivoDestino);

    if (yaExiste) {
        console.warn(`[VAULT] La nota "${nombreArchivo}.md" ya existe en ${carpeta}/. No se sobreescribió.`);
        return { ruta: `${carpeta}/${nombreArchivo}.md`, creada: false };
    }

    const frontmatter = [
        '---',
        `tipo: ${tipo}`,
        `creado: ${timestamp}`,
        `fuente: IA_nota`,
        '---',
        `# ${titulo}`,
        ''
    ].join('\n');

    fs.writeFileSync(archivoDestino, frontmatter + contenido);
    console.log(`[VAULT] Nota estructurada creada: ${carpeta}/${nombreArchivo}.md`);

    return { ruta: `${carpeta}/${nombreArchivo}.md`, creada: true };
}

 * Extrae fragmentos relevantes del vault basándose en palabras clave de la pregunta.
 * @param {string} pregunta
 * @param {number} maxFragmentos
 * @returns {Array<{fuente: string, texto: string, score: number}>}
 */
function buscarContexto(pregunta, maxFragmentos = 5) {
    const notas = leerVaultRecursivo();
    if (notas.length === 0) return [];

    const stopwords = new Set([
        'que', 'los', 'las', 'del', 'una', 'uno', 'con', 'para', 'por', 'como',
        'este', 'esta', 'ese', 'esa', 'más', 'pero', 'sin', 'eso', 'son',
        'tiene', 'hay', 'ser', 'fue', 'todo', 'todos', 'puede', 'desde',
        'the', 'and', 'for', 'are', 'but', 'not', 'you', 'all', 'can',
        'qué', 'cuál', 'cómo', 'dónde', 'cuánto', 'cuándo'
    ]);

    const palabrasClave = pregunta
        .toLowerCase()
        .replace(/[^\wáéíóúüñ\s]/g, '')
        .split(/\s+/)
        .filter(p => p.length >= 3 && !stopwords.has(p));

    if (palabrasClave.length === 0) return [];

    const fragmentosPuntuados = [];

    for (const nota of notas) {
        const secciones = nota.contenido.split(/^(?=#{1,3}\s)/m);

        for (const seccion of secciones) {
            const textoLimpio = seccion.trim();
            if (textoLimpio.length < 10) continue;

            const textoLower = textoLimpio.toLowerCase();
            const nombreLower = nota.nombre.toLowerCase();

            let score = 0;
            for (const palabra of palabrasClave) {
                if (nombreLower.includes(palabra)) score += 15;
                if (textoLower.includes(palabra)) {
                    const ocurrencias = (textoLower.match(new RegExp(palabra, 'g')) || []).length;
                    score += Math.min(ocurrencias, 3);
                }
            }

            if (palabrasClave.length > 1) {
                const fraseExacta = palabrasClave.join(' ');
                if (textoLower.includes(fraseExacta)) {
                    score += 20;
                } else {
                    for (let i = 0; i < palabrasClave.length - 1; i++) {
                        const par = palabrasClave[i] + ' ' + palabrasClave[i + 1];
                        if (textoLower.includes(par)) score += 10;
                    }
                }
            }

            if (score > 0) {
                fragmentosPuntuados.push({
                    fuente: `${nota.nombre} (${nota.ruta})`,
                    texto: textoLimpio.substring(0, 800),
                    score
                });
            }
        }
    }

    fragmentosPuntuados.sort((a, b) => b.score - a.score);
    return fragmentosPuntuados.slice(0, maxFragmentos);
}

/**
 * Genera el JSON de nodos y enlaces para el grafo 3D.
 */
function generarGrafoJSON() {
    const notas = leerVaultRecursivo();
    const nodes = [];
    const links = [];
    const regexEnlaces = /\[\[(.*?)\]\]/g;

    for (const nota of notas) {
        let dir = path.dirname(nota.ruta).replace(/\\/g, '/');
        let grupo = (dir === '.' || dir === '/' || dir === '') ? 'Cerebro/Raíz' : dir.replace(/^\//, '');

        nodes.push({ id: nota.nombre, grupo });

        let match;
        while ((match = regexEnlaces.exec(nota.contenido)) !== null) {
            const targetReal = match[1].split('|')[0].trim();
            links.push({ source: nota.nombre, target: targetReal });
        }
    }

    return { nodes, links };
}

/**
 * Estadísticas del grafo: hubs, huérfanos y conteo por carpeta.
 * Endpoint: GET /api/grafo/stats
 */
function obtenerEstadisticasGrafo() {
    const { nodes, links } = generarGrafoJSON();

    // Contar conexiones entrantes por nodo
    const contadorEntrantes = {};
    for (const node of nodes) contadorEntrantes[node.id] = 0;
    for (const link of links) {
        if (contadorEntrantes[link.target] !== undefined) {
            contadorEntrantes[link.target]++;
        }
    }

    // Contar conexiones salientes por nodo
    const contadorSalientes = {};
    for (const node of nodes) contadorSalientes[node.id] = 0;
    for (const link of links) {
        if (contadorSalientes[link.source] !== undefined) {
            contadorSalientes[link.source]++;
        }
    }

    // Nodos huérfanos: 0 entrantes Y 0 salientes
    const huerfanos = nodes
        .filter(n => contadorEntrantes[n.id] === 0 && contadorSalientes[n.id] === 0)
        .map(n => n.id);

    // Top 5 hubs (más conexiones entrantes)
    const hubs = nodes
        .map(n => ({ id: n.id, entrantes: contadorEntrantes[n.id], salientes: contadorSalientes[n.id] }))
        .sort((a, b) => b.entrantes - a.entrantes)
        .slice(0, 5);

    // Nodos por carpeta/grupo
    const porCarpeta = {};
    for (const node of nodes) {
        porCarpeta[node.grupo] = (porCarpeta[node.grupo] || 0) + 1;
    }

    return {
        total_nodos: nodes.length,
        total_enlaces: links.length,
        hubs,
        huerfanos,
        por_carpeta: porCarpeta
    };
}

/**

 * Usa la zona horaria REAL del sistema operativo, sin hardcodear.
 */

/**
 * Obtiene la fecha y hora actual del servidor en formato legible.
 * @returns {string} - Fecha/hora formateada con zona horaria del servidor
 */
function obtenerFechaHoraLocal() {
    const ahora = new Date();
    return ahora.toLocaleString('es-CO', {
        timeZone: Intl.DateTimeFormat().resolvedOptions().timeZone,
        weekday: 'long',
        year: 'numeric',
        month: 'long',
        day: 'numeric',
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
        hour12: true
    });
}

/**
 * Obtiene un timestamp simple para entradas del vault.
 * @returns {string} - Fecha/hora en formato corto colombiano
 */
function obtenerTimestamp() {
    return new Date().toLocaleString('es-CO');
}

module.exports = {
    obtenerFechaHoraLocal,
    obtenerTimestamp
};
